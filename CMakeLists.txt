cmake_minimum_required(VERSION 3.16)
project(ChronoFlow VERSION 0.4.0 LANGUAGES C CXX)
option(CF_BUILD_TESTS "Build SDK tests" ON)
option(CF_ENABLE_VULKAN "Build optional Vulkan backend" OFF)
add_library(chronoflow STATIC src/chronoflow.cpp src/reprojection.cpp)
target_include_directories(chronoflow PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(chronoflow PUBLIC cxx_std_17)
set_target_properties(chronoflow PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(CF_ENABLE_VULKAN)
  find_package(Vulkan REQUIRED)
  add_library(chronoflow_vulkan STATIC src/vulkan/vulkan_bridge.cpp)
  target_include_directories(chronoflow_vulkan PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
  target_compile_features(chronoflow_vulkan PUBLIC cxx_std_17)
  target_link_libraries(chronoflow_vulkan PUBLIC Vulkan::Vulkan)
  set_target_properties(chronoflow_vulkan PROPERTIES POSITION_INDEPENDENT_CODE ON)
  find_program(CF_GLSLANG_VALIDATOR glslangValidator)
  if(CF_GLSLANG_VALIDATOR)
    add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/interpolate.comp.spv
      COMMAND ${CF_GLSLANG_VALIDATOR} -V --target-env vulkan1.0
          ${CMAKE_CURRENT_SOURCE_DIR}/shaders/interpolate.comp
          -o ${CMAKE_CURRENT_BINARY_DIR}/interpolate.comp.spv
      DEPENDS shaders/interpolate.comp VERBATIM)
    add_custom_target(chronoflow_spirv ALL DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/interpolate.comp.spv)
  else()
    message(WARNING "glslangValidator not found: host must compile shaders/interpolate.comp itself")
  endif()
endif()
if(CF_BUILD_TESTS)
  enable_testing()
  add_executable(cf_tests tests/test_sdk.cpp)
  target_link_libraries(cf_tests PRIVATE chronoflow)
  add_test(NAME cf_tests COMMAND cf_tests)
  add_executable(cf_example examples/integration_example.c)
  target_link_libraries(cf_example PRIVATE chronoflow)
  set_target_properties(cf_example PROPERTIES LINKER_LANGUAGE CXX)
  add_test(NAME cf_example COMMAND cf_example)
  add_executable(cf_reprojection_tests tests/test_reprojection.cpp)
  target_link_libraries(cf_reprojection_tests PRIVATE chronoflow)
  add_test(NAME cf_reprojection_tests COMMAND cf_reprojection_tests)
endif()
